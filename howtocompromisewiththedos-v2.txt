import java.io.IOException;
import java.io.OutputStream;
import java.util.List;

public class Backend2Facade {

    private final TransactionService transactionService;
    private final ReportService reportService;
    private final UserManager userManager;

    public Backend2Facade() {
        this.transactionService = new TransactionService();
        this.reportService = new ReportService();
        this.userManager = new UserManager();
    }

    // ====== API TEXT-BASED UNTUK BACKEND 1 (via string command) ======

    public String handleTextCommand(String line) {
        if (line == null || line.isEmpty()) {
            return "ERROR|EMPTY_COMMAND";
        }

        // --- REGISTER ---
        // Format: REGISTER|username|password
        if (line.startsWith("REGISTER|")) {
            String[] parts = line.split("\\|");
            if (parts.length != 3) {
                return "ERROR|REGISTER_FORMAT";
            }
            String username = parts[1];
            String password = parts[2];

            boolean ok = userManager.registerUser(username, password);
            if (ok) {
                return "REGISTER_OK|" + username;
            } else {
                return "ERROR|USER_EXISTS|" + username;
            }
        }

        // --- LOGIN ---
        // Format: LOGIN|username|password
        if (line.startsWith("LOGIN|")) {
            String[] parts = line.split("\\|");
            if (parts.length != 3) {
                return "ERROR|LOGIN_FORMAT";
            }
            String username = parts[1];
            String password = parts[2];

            boolean valid = userManager.validateLogin(username, password);
            if (valid) {
                return "LOGIN_OK|" + username;
            } else {
                return "LOGIN_FAIL|INVALID_CREDENTIALS";
            }
        }

        // --- CHANGE PASSWORD ---
        // Format: CHANGE_PASSWORD|username|oldPassword|newPassword
        if (line.startsWith("CHANGE_PASSWORD|")) {
            String[] parts = line.split("\\|");
            if (parts.length != 4) {
                return "ERROR|CHANGE_PASSWORD_FORMAT";
            }
            String username    = parts[1];
            String oldPassword = parts[2];
            String newPassword = parts[3];

            boolean ok = userManager.changePassword(username, oldPassword, newPassword);
            if (ok) {
                return "CHANGE_PASSWORD_OK|" + username;
            } else {
                return "CHANGE_PASSWORD_FAIL|INVALID_CREDENTIALS";
            }
        }

        // --- ADD transaksi ---
        // Format: ADD|username|date|description|category|type|amount
        if (line.startsWith("ADD|")) {
            transactionService.addTransactionFromCommand(line);
            FinancialSummary summary = transactionService.getSummary();
            return summary.toProtocolString(); // SUMMARY|...
        }

        // --- UPDATE transaksi ---
        // Format: UPDATE|id|username|date|description|category|type|amount
        if (line.startsWith("UPDATE|")) {
            boolean updated = transactionService.updateTransactionFromCommand(line);
            if (updated) {
                FinancialSummary summary = transactionService.getSummary();
                return summary.toProtocolString();
            } else {
                return "ERROR|UPDATE_NOT_FOUND";
            }
        }

        // --- DELETE transaksi ---
        // Format: DELETE|id
        if (line.startsWith("DELETE|")) {
            boolean deleted = transactionService.deleteTransactionFromCommand(line);
            if (deleted) {
                FinancialSummary summary = transactionService.getSummary();
                return summary.toProtocolString();
            } else {
                return "ERROR|DELETE_NOT_FOUND";
            }
        }

        // --- GET_ALL transaksi ---
        if ("GET_ALL".equalsIgnoreCase(line)) {
            return transactionService.buildAllTransactionsMessage();
        }

        // --- GET_SUMMARY untuk dashboard ---
        if ("GET_SUMMARY".equalsIgnoreCase(line)) {
            FinancialSummary summary = transactionService.getSummary();
            return summary.toProtocolString();
        }

        // --- REQ_REPORT: balas header FILE_SIZE ---
        if ("REQ_REPORT".equalsIgnoreCase(line)) {
            byte[] report = getReportBytes();
            int size = report.length;
            return "FILE_SIZE|" + size;
        }

        return "ERROR|UNKNOWN_COMMAND|" + line;
    }

    // ====== API BINARY UNTUK LAPORAN ======

    public byte[] getReportBytes() {
        List<Transaction> list = transactionService.getAllTransactions();
        return reportService.buildCsvBytes(list);
    }

    public void writeReportToStream(OutputStream out) throws IOException {
        byte[] data = getReportBytes();
        out.write(data);
        out.flush();
    }

    public TransactionService getTransactionService() {
        return transactionService;
    }
}

import java.io.Serializable;

public class FinancialSummary implements Serializable {

    private double totalBalance;
    private double totalIncome;
    private double totalExpense;

    public FinancialSummary(double totalIncome, double totalExpense) {
        this.totalIncome = totalIncome;
        this.totalExpense = totalExpense;
        this.totalBalance = totalIncome - totalExpense;
    }

    public double getTotalBalance() {
        return totalBalance;
    }

    public double getTotalIncome() {
        return totalIncome;
    }

    public double getTotalExpense() {
        return totalExpense;
    }

    /**
     * Dipakai untuk kirim ke client lewat socket.
     * Angka dibulatkan ke long supaya tidak keluar dalam format 1.7E7.
     *
     * Format: SUMMARY|balance|income|expense
     */
    public String toProtocolString() {
        long balance = Math.round(totalBalance);
        long income  = Math.round(totalIncome);
        long expense = Math.round(totalExpense);

        return "SUMMARY|" + balance + "|" + income + "|" + expense;
    }
}

import java.io.Serializable;

public class Transaction implements Serializable {

    private String id;          // ID unik (UUID / timestamp)
    private String username;    // siapa yang input
    private String date;        // format: YYYY-MM-DD
    private String description; // keterangan
    private String category;    // contoh: Gaji, Makanan, Transport
    private String type;        // "INCOME" atau "EXPENSE"
    private double amount;      // nominal

    public Transaction(String id, String username, String date,
                       String description, String category,
                       String type, double amount) {
        this.id = id;
        this.username = username;
        this.date = date;
        this.description = description;
        this.category = category;
        this.type = type;
        this.amount = amount;
    }

    // Dipakai saat simpan ke file (Stream)
    public String toCSV() {
        return id + "|" + username + "|" + date + "|" + description + "|" 
               + category + "|" + type + "|" + amount;
    }

    // Dipakai saat baca dari file
    public static Transaction fromCSV(String line) {
        String[] parts = line.split("\\|");
        if (parts.length != 7) {
            throw new IllegalArgumentException("Format data tidak valid: " + line);
        }
        return new Transaction(
                parts[0],                       // id
                parts[1],                       // username
                parts[2],                       // date
                parts[3],                       // description
                parts[4],                       // category
                parts[5],                       // type
                Double.parseDouble(parts[6])    // amount
        );
    }

    public String getId() {
        return id;
    }

    public String getUsername() {
        return username;
    }

    public String getDate() {
        return date;
    }

    public String getDescription() {
        return description;
    }

    }

    public String getType() {

import java.io.*;
import java.util.ArrayList;
import java.util.List;

public class FileManager {

    private static final String FILE_NAME = "data/database_transactions.txt";

    public List<Transaction> loadTransactions() {
        List<Transaction> list = new ArrayList<>();
        File file = new File(FILE_NAME);

        if (!file.exists()) {
            return list;
        }

        try (BufferedReader reader = new BufferedReader(
                new InputStreamReader(new FileInputStream(file), "UTF-8"))) {

            String line;
            while ((line = reader.readLine()) != null) {
                if (!line.trim().isEmpty()) {
                    Transaction t = Transaction.fromCSV(line);
                    list.add(t);
                }
            }
        } catch (IOException e) {
            e.printStackTrace();
        }

        return list;
    }

    public void saveTransaction(Transaction t) {
        try (BufferedWriter writer = new BufferedWriter(
                new OutputStreamWriter(new FileOutputStream(FILE_NAME, true), "UTF-8"))) {

            writer.write(t.toCSV());
            writer.newLine();
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    public void saveAll(List<Transaction> transactions) {
        try (BufferedWriter writer = new BufferedWriter(
                new OutputStreamWriter(new FileOutputStream(FILE_NAME, false), "UTF-8"))) {

            for (Transaction t : transactions) {
                writer.write(t.toCSV());
                writer.newLine();
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}

    // Helper kecil untuk ambil ID transaksi pertama dari respon DATA_ALL
    // Format: DATA_ALL|id;date;description;category;type;amount#id2;...
    private static String extractFirstTransactionId(String dataAllResponse) {
        if (dataAllResponse == null || !dataAllResponse.startsWith("DATA_ALL|")) {
            return null;
        }
        String payload = dataAllResponse.substring("DATA_ALL|".length());
        if (payload.isEmpty()) return null;

        String firstBlock;
        int hashIndex = payload.indexOf('#');
        if (hashIndex >= 0) {
            firstBlock = payload.substring(0, hashIndex);
        } else {
            firstBlock = payload;
        }

        int semicolonIndex = firstBlock.indexOf(';');
        if (semicolonIndex <= 0) {
            return null;
        }
        return firstBlock.substring(0, semicolonIndex);
    }

import java.nio.charset.StandardCharsets;
import java.util.List;

public class ReportService {

    public String buildCsv(List<Transaction> transactions) {
        StringBuilder sb = new StringBuilder();
        sb.append("ID,USERNAME,DATE,DESCRIPTION,CATEGORY,TYPE,AMOUNT\n");

        for (Transaction t : transactions) {
            sb.append(t.getId()).append(",")
              .append(nullSafe(t.getUsername())).append(",")
              .append(nullSafe(t.getDate())).append(",")
              .append(escapeCsv(nullSafe(t.getDescription()))).append(",")
              .append(escapeCsv(nullSafe(t.getCategory()))).append(",")
              .append(nullSafe(t.getType())).append(",")
              .append(t.getAmount())
              .append("\n");
        }

        return sb.toString();
    }

    public byte[] buildCsvBytes(List<Transaction> transactions) {
        String csv = buildCsv(transactions);
        return csv.getBytes(StandardCharsets.UTF_8);
    }

    private String escapeCsv(String text) {
        if (text.contains(",") || text.contains("\"") || text.contains("\n")) {
            return "\"" + text.replace("\"", "\"\"") + "\"";
        }
        return text;
    }

    private String nullSafe(String text) {
        return text == null ? "" : text;
    }
}

import java.util.ArrayList;
import java.util.List;
import java.util.UUID;

public class TransactionService {

    private final FileManager fileManager;
    private final List<Transaction> transactions;

    public TransactionService() {
        this.fileManager = new FileManager();
        this.transactions = new ArrayList<>(fileManager.loadTransactions());
    }

    // ====== CREATE ======

    public synchronized Transaction addTransaction(String username, String date,
                                                   String description, String category,
                                                   String type, double amount) {

        String id = UUID.randomUUID().toString();
        Transaction t = new Transaction(id, username, date, description, category, type, amount);

        transactions.add(t);
        fileManager.saveTransaction(t);

        return t;
    }

    // Format command: ADD|username|date|description|category|type|amount
    public synchronized Transaction addTransactionFromCommand(String commandLine) {
        String[] parts = commandLine.split("\\|");
        if (parts.length != 7) {
            throw new IllegalArgumentException("Format ADD tidak valid: " + commandLine);
        }

        String username    = parts[1];
        String date        = parts[2];
        String description = parts[3];
        String category    = parts[4];
        String type        = parts[5];
        double amount      = Double.parseDouble(parts[6]);

        return addTransaction(username, date, description, category, type, amount);
    }

    // ====== READ ======

    public synchronized List<Transaction> getAllTransactions() {
        return new ArrayList<>(transactions);
    }

    public synchronized FinancialSummary getSummary() {
        double income = 0.0;
        double expense = 0.0;

        for (Transaction t : transactions) {
            if ("INCOME".equalsIgnoreCase(t.getType())) {
                income += t.getAmount();
            } else if ("EXPENSE".equalsIgnoreCase(t.getType())) {
                expense += t.getAmount();
            }
        }

        return new FinancialSummary(income, expense);
    }

    // Format: DATA_ALL|id;date;description;category;type;amount#id2;...
    public synchronized String buildAllTransactionsMessage() {
        StringBuilder sb = new StringBuilder();
        sb.append("DATA_ALL|");

        for (int i = 0; i < transactions.size(); i++) {
            Transaction t = transactions.get(i);
            sb.append(t.getId()).append(";")
              .append(t.getDate()).append(";")
              .append(t.getDescription()).append(";")
              .append(t.getCategory()).append(";")
              .append(t.getType()).append(";")
              .append(t.getAmount());

            if (i < transactions.size() - 1) {
                sb.append("#");
            }
        }

        return sb.toString();
    }

    // ====== UPDATE ======

    // Format command: UPDATE|id|username|date|description|category|type|amount
    public synchronized boolean updateTransactionFromCommand(String commandLine) {
        String[] parts = commandLine.split("\\|");
        if (parts.length != 8) {
            throw new IllegalArgumentException("Format UPDATE tidak valid: " + commandLine);
        }

        String id          = parts[1];
        String username    = parts[2];
        String date        = parts[3];
        String description = parts[4];
        String category    = parts[5];
        String type        = parts[6];
        double amount      = Double.parseDouble(parts[7]);

        return updateTransaction(id, username, date, description, category, type, amount);
    }

    public synchronized boolean updateTransaction(String id,
                                                  String username,
                                                  String date,
                                                  String description,
                                                  String category,
                                                  String type,
                                                  double amount) {
        for (Transaction t : transactions) {
            if (t.getId().equals(id)) {
                t.setUsername(username);
                t.setDate(date);
                t.setDescription(description);
                t.setCategory(category);
                t.setType(type);
                t.setAmount(amount);

                fileManager.saveAll(transactions);
                return true;
            }
        }
        return false;
    }

    // ====== DELETE ======

    // Format command: DELETE|id
    public synchronized boolean deleteTransactionFromCommand(String commandLine) {
        String[] parts = commandLine.split("\\|");
        if (parts.length != 2) {
            throw new IllegalArgumentException("Format DELETE tidak valid: " + commandLine);
        }
        String id = parts[1];
        return deleteTransaction(id);
    }

    public synchronized boolean deleteTransaction(String id) {
        boolean removed = transactions.removeIf(t -> t.getId().equals(id));
        if (removed) {
            fileManager.saveAll(transactions);
        }
        return removed;
    }
}

import java.io.*;
import java.util.HashMap;
import java.util.Map;

public class UserManager {

    private static final String FILE_NAME = "data/users.txt";

    private final Map<String, User> users = new HashMap<>();

    public UserManager() {
        loadUsersFromFile();
    }

    private void loadUsersFromFile() {
        File file = new File(FILE_NAME);
        if (!file.exists()) {
            return;
        }

        try (BufferedReader reader = new BufferedReader(
                new InputStreamReader(new FileInputStream(file), "UTF-8"))) {

            String line;
            while ((line = reader.readLine()) != null) {
                if (!line.trim().isEmpty()) {
                    User u = User.fromCSV(line);
                    users.put(u.getUsername(), u);
                }
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    // ====== CREATE ======

    public synchronized boolean registerUser(String username, String password) {
        if (users.containsKey(username)) {
            return false;
        }

        User u = new User(username, password);
        users.put(username, u);

        try (BufferedWriter writer = new BufferedWriter(
                new OutputStreamWriter(new FileOutputStream(FILE_NAME, true), "UTF-8"))) {

            writer.write(u.toCSV());
            writer.newLine();
        } catch (IOException e) {
            e.printStackTrace();
        }

        return true;
    }

    // ====== READ ======

    public synchronized boolean validateLogin(String username, String password) {
        User u = users.get(username);
        if (u == null) return false;
        return u.getPassword().equals(password);
    }

    public synchronized User getUser(String username) {
        return users.get(username);
    }

    // ====== UPDATE ======

    public synchronized boolean changePassword(String username,
                                               String oldPassword,
                                               String newPassword) {
        User u = users.get(username);
        if (u == null) {
            return false;
        }
        if (!u.getPassword().equals(oldPassword)) {
            return false;
        }

        User updated = new User(username, newPassword);
        users.put(username, updated);

        saveAllUsersToFile();
        return true;
    }

    private void saveAllUsersToFile() {
        try (BufferedWriter writer = new BufferedWriter(
                new OutputStreamWriter(new FileOutputStream(FILE_NAME, false), "UTF-8"))) {

            for (User u : users.values()) {
                writer.write(u.toCSV());
                writer.newLine();
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
